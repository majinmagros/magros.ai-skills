### Peer Docker Compose

```yaml
# docker-compose-peers.yaml
services:
  peer0.org1.example.com:
    image: hyperledger/fabric-peer:2.5
    container_name: peer0.org1.example.com
    environment:
      # Core peer config
      - CORE_PEER_ID=peer0.org1.example.com
      - CORE_PEER_ADDRESS=peer0.org1.example.com:7051
      - CORE_PEER_LOCALMSPID=Org1MSP
      - CORE_PEER_MSPCONFIGPATH=/etc/hyperledger/fabric/msp
      
      # TLS
      - CORE_PEER_TLS_ENABLED=true
      - CORE_PEER_TLS_CERT_FILE=/etc/hyperledger/fabric/tls/server.crt
      - CORE_PEER_TLS_KEY_FILE=/etc/hyperledger/fabric/tls/server.key
      - CORE_PEER_TLS_ROOTCERT_FILE=/etc/hyperledger/fabric/tls/ca.crt
      - CORE_PEER_TLS_CLIENTAUTHREQUIRED=true
      - CORE_PEER_TLS_CLIENTCERT_FILE=/etc/hyperledger/fabric/tls/server.crt
      - CORE_PEER_TLS_CLIENTKEY_FILE=/etc/hyperledger/fabric/tls/server.key
      - CORE_PEER_TLS_CLIENTROOTCAS_FILES=/etc/hyperledger/fabric/tls/ca.crt
      
      # Gossip
      - CORE_PEER_GOSSIP_USELEADERELECTION=true
      - CORE_PEER_GOSSIP_ORGLEADER=false
      - CORE_PEER_GOSSIP_EXTERNALENDPOINT=peer0.org1.example.com:7051
      - CORE_PEER_GOSSIP_BOOTSTRAP=peer1.org1.example.com:8051
      
      # Chaincode
      - CORE_CHAINCODE_LISTENADDRESS=0.0.0.0:7052
      - CORE_CHAINCODE_LOGLEVEL=INFO
      
      # VM
      - CORE_VM_ENDPOINT=unix:///host/var/run/docker.sock
      - CORE_VM_DOCKER_HOSTCONFIG_NETWORKMODE=fabric_network
      
      # Logging
      - CORE_LOGGING_LEVEL=INFO
      - CORE_LOGGING_FORMAT=json
      
      # CouchDB
      - CORE_LEDGER_STATE_STATEDATABASE=CouchDB
      - CORE_LEDGER_STATE_COUCHDBCONFIG_COUCHDBADDRESS=couchdb0:5984
      - CORE_LEDGER_STATE_COUCHDBCONFIG_USERNAME=admin
      - CORE_LEDGER_STATE_COUCHDBCONFIG_PASSWORD=adminpw
      
    volumes:
      - ./crypto-config/peerOrganizations/org1.example.com/peers/peer0.org1.example.com:/etc/hyperledger/fabric
      - ./channel-artifacts:/etc/hyperledger/configtx
      - peer0.org1.example.com:/var/hyperledger/production
    ports:
      - "7051:7051"
      - "7052:7052"
      - "7053:7053"
    networks:
      - fabric_network
    depends_on:
      - couchdb0

  couchdb0:
    image: couchdb:3.2
    container_name: couchdb0
    environment:
      - COUCHDB_USER=admin
      - COUCHDB_PASSWORD=adminpw
    ports:
      - "5984:5984"
    networks:
      - fabric_network
    volumes:
      - couchdb0:/opt/couchdb/data

  peer1.org1.example.com:
    # Similar config, different ports (8051, 8052, 8053)
    # ...
```

### Peer Lifecycle Commands

```bash
# Join channel
peer channel join -b mychannel.block

# Install chaincode
peer lifecycle chaincode install assettransfer.tar.gz

# Query installed
peer lifecycle chaincode queryinstalled

# Approve chaincode
peer lifecycle chaincode approveformyorg \
  -o orderer.example.com:7050 \
  --channelID mychannel \
  --name assettransfer \
  --version 1.0 \
  --package-id $(peer lifecycle chaincode queryinstalled | grep assettransfer_1.0 | awk '{print $1}') \
  --sequence 1 \
  --tls \
  --cafile $ORDERER_CA

# Check commit readiness
peer lifecycle chaincode checkcommitreadiness \
  --channelID mychannel \
  --name assettransfer \
  --version 1.0 \
  --sequence 1 \
  --tls \
  --cafile $ORDERER_CA \
  --output json

# Commit chaincode
peer lifecycle chaincode commit \
  -o orderer.example.com:7050 \
  --channelID mychannel \
  --name assettransfer \
  --version 1.0 \
  --sequence 1 \
  --tls \
  --cafile $ORDERER_CA \
  --peerAddresses peer0.org1.example.com:7051 \
  --tlsRootCertFiles $PEER0_ORG1_CA \
  --peerAddresses peer0.org2.example.com:9051 \
  --tlsRootCertFiles $PEER0_ORG2_CA

# Invoke transaction
peer chaincode invoke \
  -o orderer.example.com:7050 \
  --channelID mychannel \
  --name assettransfer \
  --peerAddresses peer0.org1.example.com:7051 \
  --tlsRootCertFiles $PEER0_ORG1_CA \
  --peerAddresses peer0.org2.example.com:9051 \
  --tlsRootCertFiles $PEER0_ORG2_CA \
  --isInit \
  -c '{"function":"CreateAsset","Args":["asset1","blue","5","Org1", "300"]}'

# Query
peer chaincode query -C mychannel -n assettransfer -c '{"function":"ReadAsset","Args":["asset1"]}'
```

### Environment Variables Reference

| Variable | Description | Default |
|----------|-------------|---------|
| `CORE_PEER_ID` | Peer ID | peer0.org1.example.com |
| `CORE_PEER_ADDRESS` | Listen address | peer0.org1.example.com:7051 |
| `CORE_PEER_LOCALMSPID` | MSP ID | Org1MSP |
| `CORE_PEER_MSPCONFIGPATH` | MSP config path | /etc/hyperledger/fabric/msp |
| `CORE_PEER_TLS_ENABLED` | Enable TLS | true |
| `CORE_PEER_TLS_CERT_FILE` | TLS cert | /etc/hyperledger/fabric/tls/server.crt |
| `CORE_PEER_TLS_KEY_FILE` | TLS key | /etc/hyperledger/fabric/tls/server.key |
| `CORE_PEER_TLS_ROOTCERT_FILE` | TLS CA cert | /etc/hyperledger/fabric/tls/ca.crt |
| `CORE_PEER_GOSSIP_USELEADERELECTION` | Leader election | true |
| `CORE_PEER_GOSSIP_ORGLEADER` | Is org leader | false |
| `CORE_PEER_GOSSIP_EXTERNALENDPOINT` | External endpoint | peer0.org1.example.com:7051 |
| `CORE_PEER_GOSSIP_BOOTSTRAP` | Bootstrap peers | peer1.org1.example.com:8051 |
| `CORE_LEDGER_STATE_STATEDATABASE` | State DB | CouchDB |
| `CORE_LEDGER_STATE_COUCHDBCONFIG_COUCHDBADDRESS` | CouchDB address | couchdb0:5984 |
